46def test_full_adaptive_session_on_a_bundled_deck() -> None: 

47 cards = load_flashcards(DATA_DIR / "python_basics.json") 

48 answers = {card.front: card.back for card in cards} 

49 session = QuizSession(QuizModeFactory.create("adaptive", cards, random.Random(4))) 

50 first_attempts: set[str] = set() 

51 

52 while (card := session.next_card()) is not None: 

53 # Miss every card on its first attempt, answer correctly afterwards. 

54 if card.front not in first_attempts: 

55 first_attempts.add(card.front) 

56 session.submit("definitely wrong") 

57 else: 

58 session.submit(answers[card.front]) 

59 

60 assert session.stats.total == 2 * len(cards) 

61 assert session.stats.correct == len(cards) 

62 assert sorted(session.stats.missed) == sorted(answers)
