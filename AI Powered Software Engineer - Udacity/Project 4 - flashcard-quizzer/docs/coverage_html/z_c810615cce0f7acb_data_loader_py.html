1"""Loading and validating flashcard decks from JSON files. 

2 

3Two layouts are supported: 

4 

5* Array format: ``[{"front": "...", "back": "..."}, ...]`` 

6* Object format: ``{"cards": [{"front": "...", "back": "..."}, ...]}`` 

7 

8Every problem is reported as a :class:`FlashcardLoadError` with a short, 

9human-readable message, so callers never have to show a raw traceback. 

10""" 

11 

12import json 

13from pathlib import Path 

14 

15from utils.models import Flashcard 

16 

17MAX_FILE_BYTES = 5_000_000 # Refuse absurdly large decks instead of loading them. 

18REQUIRED_FIELDS = ("front", "back") 

19 

20 

21class FlashcardLoadError(Exception): 

22 """Raised when a deck cannot be read or does not have a valid structure.""" 

23 

24 

25def load_flashcards(path: str | Path) -> list[Flashcard]: 

26 """Read a JSON deck from disk and return its validated cards. 

27 

28 Args: 

29 path: Location of the JSON deck. 

30 

31 Returns: 

32 The cards in file order. 

33 

34 Raises: 

35 FlashcardLoadError: If the file is missing, unreadable, too large, 

36 not valid JSON, or contains an invalid card. 

37 """ 

38 deck_path = Path(path) 

39 text = _read_text(deck_path) 

40 try: 

41 data: object = json.loads(text) 

42 except json.JSONDecodeError as exc: 

43 raise FlashcardLoadError( 

44 f"Invalid JSON in {deck_path}: {exc.msg} " 

45 f"(line {exc.lineno}, column {exc.colno})." 

46 ) from exc 

47 return parse_flashcards(data) 

48 

49 

50def parse_flashcards(data: object) -> list[Flashcard]: 

51 """Validate already-parsed JSON data and convert it to flashcards. 

52 

53 Args: 

54 data: The result of ``json.loads`` for a deck. 

55 

56 Returns: 

57 The validated cards, with surrounding whitespace trimmed. 

58 

59 Raises: 

60 FlashcardLoadError: If the structure is wrong or a card is invalid. 

61 """ 

62 raw_cards = _extract_card_list(data) 

63 if not raw_cards: 

64 raise FlashcardLoadError("The deck contains no cards.") 

65 return [_parse_card(item, index) for index, item in enumerate(raw_cards, 1)] 

66 

67 

68def _read_text(path: Path) -> str: 

69 """Read a deck file, translating OS errors into friendly messages.""" 

70 try: 

71 if path.stat().st_size > MAX_FILE_BYTES: 

72 raise FlashcardLoadError( 

73 f"{path} is too large (limit is {MAX_FILE_BYTES:,} bytes)." 

74 ) 

75 return path.read_text(encoding="utf-8") 

76 except FileNotFoundError: 

77 raise FlashcardLoadError(f"File not found: {path}") from None 

78 except IsADirectoryError: 

79 raise FlashcardLoadError(f"{path} is a directory, not a JSON file.") from None 

80 except PermissionError: 

81 raise FlashcardLoadError(f"Permission denied when reading {path}.") from None 

82 except UnicodeDecodeError: 

83 raise FlashcardLoadError(f"{path} is not valid UTF-8 text.") from None 

84 except OSError as exc: 

85 raise FlashcardLoadError(f"Could not read {path}: {exc.strerror}.") from exc 

86 

87 

88def _extract_card_list(data: object) -> list[object]: 

89 """Return the list of raw cards from either supported layout.""" 

90 if isinstance(data, list): 

91 return data 

92 if isinstance(data, dict): 

93 if "cards" not in data: 

94 raise FlashcardLoadError('Object format needs a "cards" list.') 

95 cards = data["cards"] 

96 if not isinstance(cards, list): 

97 raise FlashcardLoadError('The "cards" value must be a list.') 

98 return cards 

99 raise FlashcardLoadError( 

100 'The deck must be a JSON array or an object with a "cards" list.' 

101 ) 

102 

103 

104def _parse_card(item: object, index: int) -> Flashcard: 

105 """Validate one raw card; ``index`` is 1-based for error messages.""" 

106 if not isinstance(item, dict): 

107 raise FlashcardLoadError( 

108 f'Card {index}: expected an object with "front" and "back".' 

109 ) 

110 values: dict[str, str] = {} 

111 for field_name in REQUIRED_FIELDS: 

112 value = item.get(field_name) 

113 if not isinstance(value, str) or not value.strip(): 

114 raise FlashcardLoadError( 

115 f'Card {index}: missing or empty "{field_name}" text.' 

116 ) 

117 values[field_name] = value.strip() 

118 return Flashcard(front=values["front"], back=values["back"])
