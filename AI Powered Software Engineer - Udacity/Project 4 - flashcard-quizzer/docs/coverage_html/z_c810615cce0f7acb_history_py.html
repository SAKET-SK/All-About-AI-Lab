1"""Persistent session history, used for the ``--stats`` lifetime statistics. 

2 

3History is stored as a JSON list in the user's home directory. The store takes 

4its path as an argument, so tests and callers can point it anywhere. 

5""" 

6 

7import json 

8from collections import Counter 

9from dataclasses import asdict, dataclass, field 

10from datetime import datetime, timezone 

11from pathlib import Path 

12 

13HARDEST_CARD_COUNT = 5 

14 

15 

16def default_history_path() -> Path: 

17 """Return where history is stored when no path is given. 

18 

19 This is ``~/.flashcard_quizzer/history.json``. If the home directory cannot 

20 be determined, the current directory is used instead of failing. 

21 """ 

22 try: 

23 base = Path.home() 

24 except RuntimeError: 

25 base = Path.cwd() 

26 return base / ".flashcard_quizzer" / "history.json"

29class HistoryError(Exception): 

30 """Raised when the history file cannot be read, parsed or written.""" 

31

33@dataclass(frozen=True) 

34class SessionRecord: 

35 """One finished quiz session. 

36 

37 Attributes: 

38 deck: Identifier of the deck (the resolved file path). 

39 mode: Name of the quiz mode that was used. 

40 timestamp: UTC time the session ended, in ISO 8601 format. 

41 total: Number of answers submitted. 

42 correct: Number of correct answers. 

43 missed: Fronts of the cards missed during the session. 

44 """ 

45 

46 deck: str 

47 mode: str 

48 timestamp: str 

49 total: int 

50 correct: int 

51 missed: tuple[str, ...] = () 

52 

53 @classmethod 

54 def create( 

55 cls, deck: str, mode: str, total: int, correct: int, missed: list[str] 

56 ) -> "SessionRecord": 

57 """Build a record stamped with the current UTC time.""" 

58 stamp = datetime.now(timezone.utc).isoformat(timespec="seconds") 

59 return cls(deck, mode, stamp, total, correct, tuple(missed))

62@dataclass(frozen=True) 

63class LifetimeStats: 

64 """Totals across every saved session for one deck. 

65 

66 Attributes: 

67 sessions: Number of saved sessions. 

68 total: Total answers submitted. 

69 correct: Total correct answers. 

70 hardest: Up to five ``(front, times_missed)`` pairs, most missed first. 

71 """ 

72 

73 sessions: int = 0 

74 total: int = 0 

75 correct: int = 0 

76 hardest: list[tuple[str, int]] = field(default_factory=list) 

77 

78 @property 

79 def accuracy(self) -> float: 

80 """Return the lifetime accuracy percentage, or 0.0 with no answers.""" 

81 if self.total == 0: 

82 return 0.0 

83 return 100.0 * self.correct / self.total

86class HistoryStore: 

87 """Reads and appends :class:`SessionRecord` entries in a JSON file.""" 

88 

89 def __init__(self, path: Path | None = None) -> None: 

90 """Remember where the history file lives. 

91 

92 Args: 

93 path: Location of the JSON history file (default location if None). 

94 """ 

95 self._path = path if path is not None else default_history_path() 

96 

97 def load(self) -> list[SessionRecord]: 

98 """Return all saved records; a missing file means no history yet. 

99 

100 Raises: 

101 HistoryError: If the file exists but cannot be read or parsed. 

102 """ 

103 if not self._path.exists(): 

104 return [] 

105 try: 

106 data = json.loads(self._path.read_text(encoding="utf-8")) 

107 if not isinstance(data, list): 

108 raise ValueError("expected a JSON list") 

109 return [_record_from_dict(item) for item in data] 

110 except (OSError, ValueError, TypeError, KeyError) as exc: 

111 raise HistoryError( 

112 f"History file {self._path} is unreadable: {exc}" 

113 ) from exc 

114 

115 def append(self, record: SessionRecord) -> None: 

116 """Add a record, writing the file atomically. 

117 

118 Args: 

119 record: The session to save. 

120 

121 Raises: 

122 HistoryError: If the existing history is corrupt or cannot be 

123 written. A corrupt file is never overwritten. 

124 """ 

125 records = self.load() 

126 records.append(record) 

127 payload = json.dumps([asdict(item) for item in records], indent=2) 

128 temp_path = self._path.with_suffix(".tmp") 

129 try: 

130 self._path.parent.mkdir(parents=True, exist_ok=True) 

131 temp_path.write_text(payload, encoding="utf-8") 

132 temp_path.replace(self._path) 

133 except OSError as exc: 

134 raise HistoryError( 

135 f"Could not save history to {self._path}: {exc}" 

136 ) from exc 

137 

138 def summarize(self, deck: str) -> LifetimeStats: 

139 """Combine all saved sessions for one deck. 

140 

141 Args: 

142 deck: Deck identifier to filter on. 

143 

144 Returns: 

145 Lifetime totals and the most frequently missed cards. 

146 

147 Raises: 

148 HistoryError: If the history file is unreadable. 

149 """ 

150 records = [item for item in self.load() if item.deck == deck] 

151 misses: Counter[str] = Counter() 

152 for item in records: 

153 misses.update(item.missed) 

154 return LifetimeStats( 

155 sessions=len(records), 

156 total=sum(item.total for item in records), 

157 correct=sum(item.correct for item in records), 

158 hardest=misses.most_common(HARDEST_CARD_COUNT), 

159 ) 

160 

161 

162def _record_from_dict(item: object) -> SessionRecord: 

163 """Convert one JSON object to a record, raising on a bad shape.""" 

164 if not isinstance(item, dict): 

165 raise TypeError("each history entry must be an object") 

166 return SessionRecord( 

167 deck=str(item["deck"]), 

168 mode=str(item["mode"]), 

169 timestamp=str(item["timestamp"]), 

170 total=int(item["total"]), 

171 correct=int(item["correct"]), 

172 missed=tuple(str(front) for front in item.get("missed", [])), 

173 )
